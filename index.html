06 · Scripts

Google Sheets and everyday automation

Scripts that turn repetitive office work into a single click.

A booking tracker that watches the balances

This Apps Script adds a Bookings menu to a Google Sheet. One click:

  • works out the balance due on every booking
  • shades every overdue row
  • writes a month-by-month summary tab

A working sample written for this portfolio.

Bookings.gs
// Bookings sheet, columns A to E: Event date | Client | Package | Total | Paid
// The script writes the balance due into column F.
function onOpen() {
  SpreadsheetApp.getUi().createMenu('Bookings')
    .addItem('Update balances and summary', 'updateBookings')
    .addToUi();
}

function updateBookings() {
  const ss = SpreadsheetApp.getActive();
  const sheet = ss.getSheetByName('Bookings');
  const rows = sheet.getDataRange().getValues().slice(1);
  const today = new Date();
  const months = {};

  rows.forEach((r, i) => {
    const [date, client, pkg, total, paid] = r;
    if (!client || !(date instanceof Date)) return;
    const due = Number(total) - Number(paid);
    const row = i + 2;
    sheet.getRange(row, 6).setValue(due);
    const late = due > 0 && date < today;
    sheet.getRange(row, 1, 1, 6).setBackground(late ? '#f6d5d0' : null);
    const key = Utilities.formatDate(date, ss.getSpreadsheetTimeZone(), 'yyyy-MM');
    months[key] = months[key] || [0, 0, 0];
    months[key][0] += 1;
    months[key][1] += Number(total);
    months[key][2] += due;
  });

  const out = ss.getSheetByName('Summary') || ss.insertSheet('Summary');
  out.clearContents();
  const table = [['Month', 'Bookings', 'Booked', 'Still due']];
  Object.keys(months).sort().forEach(m => table.push([m, ...months[m]]));
  out.getRange(1, 1, table.length, 4).setValues(table);
}
